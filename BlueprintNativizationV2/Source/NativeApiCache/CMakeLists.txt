cmake_minimum_required(VERSION 3.24)
project(NativeApiCache VERSION 0.1.0 LANGUAGES CXX)
if(NOT TARGET CppSourceParser)
    add_subdirectory(../CppSourceParser CppSourceParser)
endif()
add_library(NativeApiCache STATIC Private/NativeApiCache.cpp)
target_compile_features(NativeApiCache PUBLIC cxx_std_17)
target_include_directories(NativeApiCache PUBLIC Public)
target_link_libraries(NativeApiCache PUBLIC CppSourceParser)
if(MSVC)
    target_compile_options(NativeApiCache PRIVATE /W4 /utf-8)
endif()
